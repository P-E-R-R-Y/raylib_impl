141void RayWindow::drawPoly(graphic::IPolygon *polygon) {
142 RayPolygon *rayPolygon = static_cast<RayPolygon *>(polygon);
143 // by reference : copying the vector every frame would be a per-frame
144 // allocation for nothing, the triangles are computed at construction
145 const std::vector<Triangle<double>> &triangles = rayPolygon->_triangles;
146
147 for (size_t i = 0; i < triangles.size(); i++) {
148 //raylib asks for counter clockwise, hence the reversed order
149 raylib::DrawTriangle(
150 {static_cast<float>(triangles[i].p3.x + rayPolygon->getPosition().x), static_cast<float>(triangles[i].p3.y + rayPolygon->getPosition().y)},
151 {static_cast<float>(triangles[i].p2.x + rayPolygon->getPosition().x), static_cast<float>(triangles[i].p2.y + rayPolygon->getPosition().y)},
152 {static_cast<float>(triangles[i].p1.x + rayPolygon->getPosition().x), static_cast<float>(triangles[i].p1.y + rayPolygon->getPosition().y)},
153 rayPolygon->_color);
154 }
155}

190void RayWindow::drawText3D(graphic::IText *text, Vector3f position) {
191 //raylib has no built-in DrawText3D : project the world position through
192 //the active camera and draw the same 2D text there (screen-space billboard).
193 //2D draw calls need orthographic mode, so step out of BeginMode3D and
194 //back in around the projected draw - the caller's own endMode3() closes
195 //the block they think they're still in.
196 RayText *rayText = static_cast<RayText *>(text);
197 if (!_activeCamera) return;
198
199 raylib::Vector3 worldPos = {float(position.x), float(position.y), float(position.z)};
200 raylib::Vector2 screenPos = raylib::GetWorldToScreen(worldPos, *_activeCamera);
201
202 raylib::EndMode3D();
203 raylib::DrawTextPro(rayText->_font->handle(), rayText->_data.c_str(), screenPos, {0, 0}, rayText->_rotation,
204 rayText->_size, rayText->_spacing, rayText->_color);
205 raylib::BeginMode3D(*_activeCamera);
206}

208void RayWindow::endMode3() {
